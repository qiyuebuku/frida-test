        153 def derived_signal_record(
        154     *,
        155     table: str,
        156     pk,
        157     target_ref: dict,
        158     signal_type: str,
        159     observed_at,
        160     value,
        161     unit: str | None = None,
        162     window: str | None = None,
        163     title: str | None = None,
        164     payload_extra: dict | None = None,
        165 ) -> dict | None:
        166     observed_at_text = iso(observed_at)
        167     safe_value = json_safe(value)
        168     if observed_at_text is None or safe_value is None:
        169         return None
        170     payload = {
        171         "target_ref": json_safe(target_ref),
        172         "signal_type": signal_type,
        173         "observed_at": observed_at_text,
        174         "value": safe_value,
        175         "unit": unit,
        176         "window": window,
        177         "title": title,
        178         "source_table": table,
        179         "source_pk": pk,
        180     }
        181     if payload_extra:
        182         payload.update(json_safe(payload_extra))
        183     return {
        184         "source_type": "derived_signal",
        185         "source_id": f"{table}:{pk}",
        186         "observed_at": observed_at_text,
        187         "payload": payload,
        188         "raw_text": title,
        189         "metadata": {"source_table": table, "source_pk": pk},
        190     }

        193 def news_row_to_record(row: dict) -> dict | None:
        194     published_at = iso(row.get("published_at") or row.get("cr
            eated_at"))

        196         return None
        197     category = (row.get("category") or "").lower()
        198     # 规则投影：只转换结构化字段，文本实体由 Adapter 内 LLM

            ow.get("title") or "",

            rce") or "ft_news",

            规则结果；LLM 会在 compile 阶段补充

  没跑），自然找不到任何东西。
